Comandos operacionales

Backend

cd /home/jose-gonzalez-oliva/LOCALES
source .venv/bin/activate
export APP_ENV=local
export BACKEND_BASE_URL=http://127.0.0.1:8000
export OLLAMA_BASE_URL=<model-runtime-url>
export USE_REMOTE_RAG=false
export DOCUMENTS_DB_PATH=/home/jose-gonzalez-oliva/LOCALES/DB/chunks/documents.sqlite
uvicorn app.main:app --host 127.0.0.1 --port 8000

Frontend

python -m http.server 3000 --directory frontend

Smoke /health

curl http://127.0.0.1:8000/health

Smoke /chat

curl -X POST "http://127.0.0.1:8000/chat" -H "Content-Type: application/json" -d "{\"message\":\"Que es un transformer?\",\"use_rag\":true}"

Run chat evals

curl -X POST "http://127.0.0.1:8000/api/evals/chat/run"
